import {initializeApp} from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-app.js'; import {getAuth,signInAnonymously,onAuthStateChanged} from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-auth.js'; import {getFirestore,collection,doc,onSnapshot,runTransaction} from 'https://www.gstatic.com/firebasejs/10.14.1/firebase-firestore.js'; const ui=window.calendarUI, config=window.FIREBASE_CONFIG; if(!config||!config.apiKey||config.apiKey.includes('PASTE_')) ui.toast('أكملي ملف firebase-config.js'); else { const app=initializeApp(config), auth=getAuth(app), db=getFirestore(app), rooms=['قاعة البرمجة','قاعة نماء','قاعة الابتكار']; const eventRef=id=>doc(db,'events',id); const lockRef=e=>doc(db,'roomLocks',e.date+'_'+rooms.indexOf(e.place)); const isRoom=e=>e&&rooms.includes(e.place); window.calendarCloud={ async upsert(e,old){ const uid=auth.currentUser?.uid; if(!uid)throw Error('لا يوجد اتصال'); await runTransaction(db,async tx=>{ const ref=eventRef(e.id), existing=await tx.get(ref); if(existing.exists()&&existing.data().ownerUid!==uid) throw Error('لا يمكن تعديل نشاط من جهاز آخر'); if(old&&!existing.exists()) throw Error('النشاط لم يعد موجودًا'); const prev=existing.exists()?existing.data():null; const nr=isRoom(e)?lockRef(e):null, or=isRoom(prev)?lockRef(prev):null; const ns=nr?await tx.get(nr):null, os=or&&(!nr||or.path!==nr.path)?await tx.get(or):null; if(ns?.exists()&&ns.data().eventId!==e.id) throw Error('القاعة محجوزة بالفعل في هذا اليوم'); if(or&&(!nr||or.path!==nr.path)&&os?.data().eventId===e.id) tx.delete(or); if(nr&&!ns.exists()) tx.set(nr,{eventId:e.id,ownerUid:uid}); tx.set(ref,{...e,ownerUid:uid}); }); }, async remove(e){ const uid=auth.currentUser?.uid; if(!uid)throw Error('لا يوجد اتصال'); await runTransaction(db,async tx=>{ const ref=eventRef(e.id), snap=await tx.get(ref); if(!snap.exists())return; if(snap.data().ownerUid!==uid) throw Error('لا يمكن حذف نشاط من جهاز آخر'); const old=snap.data(), lr=isRoom(old)?lockRef(old):null, ls=lr?await tx.get(lr):null; if(ls?.exists()&&ls.data().eventId===e.id) tx.delete(lr); tx.delete(ref); }); } }; onAuthStateChanged(auth,user=>{ if(!user){ ui.setUser(null); signInAnonymously(auth).catch(e=> ui.toast('تعذر الاتصال: '+e.message) ); return; } ui.setUser(user.uid); onSnapshot( collection(db,'events'), snapshot=>ui.setEntries( snapshot.docs.map(d=>({id:d.id,...d.data()})) ), e=>ui.toast('خطأ في تحميل الأنشطة: '+e.message) ); }); }
